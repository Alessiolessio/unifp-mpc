cmake_minimum_required(VERSION 3.20)
project(unifp_mpc LANGUAGES CXX)

find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(onnxruntime CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(mujoco_dir "${Python_SITEARCH}/mujoco")
find_file(mujoco_lib libmujoco.so.3.6.0 PATHS "${mujoco_dir}" NO_DEFAULT_PATH REQUIRED)

pybind11_add_module(unifp_native native/world.cpp)
target_link_libraries(unifp_native PRIVATE Threads::Threads)
pybind11_add_module(native_probe tests/native_probe.cpp)
foreach(target IN ITEMS unifp_native native_probe)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE -ffp-contract=off)
    target_include_directories(${target} PRIVATE "${mujoco_dir}/include")
    target_link_libraries(${target} PRIVATE onnxruntime::onnxruntime "${mujoco_lib}")
endforeach()
